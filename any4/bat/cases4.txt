=== o03_returncondmark
# accepted expected: a conditional of two &return calls returned
fn g(i I64) P& { return mkp&return(i) if i > 5 else mkp&return(i + 100) }
fn main() {
    l := List<P&>()
    for i in range 2 { l.Push(g(i * 11 + 1)) }
    assert churn() > 0
    for x in l { say("returncondmark " $x.n) }
}
