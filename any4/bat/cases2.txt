=== b31_paramreturnbox
# a callee returns as its value result a scoped call naming its parameter (built where the caller's argument lives); the
# caller makes an any of it in an outer list - AnyItemNew takes the unscoped outer call for new storage built there
fn g(p String&, i I64) String { return mk&p(i) }
fn main() {
    l := List<any Sh&>()
    for i in range 3 {
        t := "inner" $i
        l.Push(Lab(g(t, i)))
    }
    assert churn() > 0
    for x in l { say($x.Show()) }
}

=== b32_paramreturnchain

    t := "inner" $i
    return g(t, i)
}
fn main() {

    b := h(2)
    assert churn() > 0
    say($a " " $b)
}

=== b34_destructreturn
# a destructured scoped call's array naming the function's own local, returned

    t := "local" $i

}

=== b37_scopedvaluestruct
# a struct value with an array field of known length (no references) from a scoped call, returned
type FA struct(a Array<I64, 3>) { a }
fn mkfa(i I64) FA { return FA(Array<I64, 3>(i, i + 1, i + 2)) }
fn g(i I64) FA {
    t := "local" $i
    return mkfa&t(i)
}
fn main() {
    a := g(1)
    assert churn() > 0
    say($a.a[0])
}
