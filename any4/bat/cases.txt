=== b08_task
# a task handed a scoped call's result built where a local of the join's loop body lives
fn use(p P&) I64 {
    s := 0
    for k in range 2000 { s += churn() % 7 }
    return p.n + s - s
}
fn main() {
    total := 0
    join {
        for i in range 4 {
            t := "inner" $i
            spawn use(mkp&t(i * 11 + 1))
        }
    }
    assert churn() > 0
    say("task done")
}
