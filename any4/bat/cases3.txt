=== b38_destructassign
# a scoped call's several results assigned (not declared) to outer locals
fn mk2p(i I64) P&, P& { return P(i), P(i + 1) }
fn main() {
    a := P(0)
    b := P(0)
    for i in range 2 {
        t := "inner" $i
        a, b = mk2p&t(i * 11 + 1)
    }
    assert churn() > 0
    say("destructassign " $a.n " " $b.n)
}

=== b41_forinscoped
# a for-in over a scoped call's List result, its elements pushed outward
fn mkl(i I64) List<P&> {
    l := List<P&>()
    l.Push(P(i))
    l.Push(P(i + 1))
    return l
}
fn main() {
    out := List<P&>()
    for i in range 2 {
        t := "inner" $i
        for q in mkl&t(i * 11 + 1) { out.Push(q) }
    }
    assert churn() > 0
    for x in out { say("forin " $x.n) }
}
