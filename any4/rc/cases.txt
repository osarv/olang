=== r01_lambdabody
# RetCopy: a borrow of the returned local kept by a lambda's body (a capture stored through a captured reference) - a
# second name the flow analysis does not walk into
fn f(k mut K&, i I64) String {
    s := "abc" $i
    put := fn() { k.s = s }
    put()
    return s
}
fn probe() String {
    k := K("x")
    x := f(k, 1)
    x[0] = 'Q'
    return "k.s " $k.s " x " $x
}
G := probe()
fn main() {
    say("run: " $probe())
    say("baked: " $G)
}

=== r16_anybox
# r01's missed name through an any's box: the box holds the call's array as it is (AnyItemNew), which the lambda's
# store also names
type W trait {
    mut Bump()
}
fn (l mut Lab&) Bump() { l[0] = 'Z' }
fn f(k mut K&, i I64) String {
    s := "abc" $i
    put := fn() { k.s = s }
    put()
    return s
}
fn probe() String {
    k := K("x")
    b mut any W& = Lab(f(k, 2))
    b.Bump()
    return "k.s " $k.s
}
G := probe()
fn main() {
    say("run: " $probe())
    say("baked: " $G)
}
